import { pgTable, uuid, text, boolean, timestamp, date, integer, numeric } from "drizzle-orm/pg-core";
import { relations } from "drizzle-orm";

export const merchants = pgTable("merchants", {
  id: uuid("id").primaryKey().defaultRandom(),
  name: text("name").notNull(),
  email: text("email"),
  phone: text("phone"),
  address: text("address"),
  logoUrl: text("logo_url"),
  createdAt: timestamp("created_at").defaultNow(),
});

export const offers = pgTable("offers", {
  id: uuid("id").primaryKey().defaultRandom(),
  merchantId: uuid("merchant_id").notNull().references(() => merchants.id, { onDelete: "cascade" }),
  title: text("title").notNull(),
  description: text("description"),
  // Pricing model: either fixed price or percent discount
  type: text("type").$type<"percent"|"fixed"|"set_menu">().default("percent"),
  percentOff: integer("percent_off"), // null if not percent
  fixedPrice: numeric("fixed_price", { precision: 10, scale: 2 }), // null if not fixed
  // Validity window and scheduling
  validFrom: date("valid_from"),
  validTo: date("valid_to"),
  daysOfWeek: text("days_of_week"), // e.g. "mon,tue,wed" (optional MVP)
  active: boolean("active").default(true),
  archived: boolean("archived").default(false),
  createdAt: timestamp("created_at").defaultNow(),
});

export const users = pgTable("users", {
  id: uuid("id").primaryKey(), // link to your auth if you have it; else seed a demo user
  role: text("role").$type<"resident"|"student"|"merchant"|"admin">().default("merchant"),
  merchantId: uuid("merchant_id").references(() => merchants.id),
  email: text("email"),
  name: text("name"),
  // Staff PIN used at point of redemption (4-6 digits)
  staffPin: text("staff_pin"),
});

export const redemptions = pgTable("redemptions", {
  id: uuid("id").primaryKey().defaultRandom(),
  userId: uuid("user_id"),          // optional for merchant view in MVP
  offerId: uuid("offer_id").notNull().references(() => offers.id),
  merchantId: uuid("merchant_id").notNull().references(() => merchants.id),
  // For merchant-side redemption: either staff PIN or device ID
  staffUserId: uuid("staff_user_id"),
  verifiedAt: timestamp("verified_at").defaultNow(),
  // Accounting snapshot at redemption time
  pricingType: text("pricing_type").$type<"percent"|"fixed"|"set_menu">(),
  percentOff: integer("percent_off"),
  fixedPrice: numeric("fixed_price", { precision: 10, scale: 2 }),
  basketSubtotal: numeric("basket_subtotal", { precision: 10, scale: 2 }), // optional, if entered
  calculatedDiscount: numeric("calculated_discount", { precision: 10, scale: 2 }), // derived
  createdAt: timestamp("created_at").defaultNow(),
});

export const billingRuns = pgTable("billing_runs", {
  id: uuid("id").primaryKey().defaultRandom(),
  merchantId: uuid("merchant_id").notNull().references(() => merchants.id),
  periodStart: date("period_start").notNull(),
  periodEnd: date("period_end").notNull(),
  redemptionCount: integer("redemption_count").default(0),
  // platform fee: e.g., 5% of basket discount or per-redemption fee; we start with per-redemption fee
  feeType: text("fee_type").$type<"per_redemption"|"percent_of_discount">().default("per_redemption"),
  feeValue: numeric("fee_value", { precision: 10, scale: 2 }).default("0.50"), // e.g., £0.50 per redemption
  totalFees: numeric("total_fees", { precision: 12, scale: 2 }).default("0.00"),
  status: text("status").$type<"draft"|"pending_dd"|"collected">().default("draft"),
  createdAt: timestamp("created_at").defaultNow(),
});
